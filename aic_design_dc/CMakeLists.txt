cmake_minimum_required(VERSION 3.14)
project(hsx_core CXX)
if(MSVC)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
add_library(hsx_core SHARED core/cabinet.cpp core/layout.cpp core/export.cpp core/parts.cpp core/board.cpp core/joinery.cpp core/nest.cpp core/capi.cpp)
add_executable(test_core tests/test_core.cpp core/cabinet.cpp core/layout.cpp core/export.cpp core/parts.cpp core/board.cpp core/joinery.cpp core/nest.cpp core/capi.cpp)
find_package(Threads REQUIRED)
target_link_libraries(hsx_core PRIVATE Threads::Threads)
target_link_libraries(test_core PRIVATE Threads::Threads)
enable_testing()
add_test(NAME core COMMAND test_core)
